//Health Bonus Pulse
event HudHealthBonusPulse
{
	Animate	PlayerStatusHealthValue			FgColor	"HP Buff" Accel 0.0 0.0
	Animate 	PlayerStatusHealthValueSpec 		FgColor 	"HP Buff" Accel 0.0 0.0
	Animate 	PlayerStatusHealthvalueTargetID 	FgColor 	"HP Buff" linear 0.0 0.0

	RunEvent HudHealthBonusPulseLoop 0.4
}

// call to loop HudHealthBonusPulse
event HudHealthBonusPulseLoop
{
	RunEvent HudHealthBonusPulse 0.0
}

event HudHealthBonusPulseStop
{
	StopEvent HudHealthBonusPulse 0.0
	StopEvent HudHealthBonusPulseLoop 0.0

	Animate	PlayerStatusHealthValue			FgColor	"HP"	Accel 0.0 0.0
	Animate 	PlayerStatusHealthValueSpec 		FgColor 	"HP" Accel 0.0 0.0
	Animate 	PlayerStatusHealthvalueTargetID 	FgColor 	"HP" linear 0.0 0.0
}
