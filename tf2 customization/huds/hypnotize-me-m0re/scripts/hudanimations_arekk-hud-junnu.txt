	Animate PlayerStatusHealthValueShadow FgColor "0 120 240 255" linear 0.0 0.0
	Animate PlayerStatusHealthValueShadowExtra Alpha 255 linear 0.0 0.0
	Animate PlayerStatusHealthValueSpecTargetShadow FgColor "0 120 240 255" linear 0.0 0.0
	Animate PlayerStatusHealthValueSpecTargetShadowExtra Alpha 255 linear 0.0 0.0
	Animate PlayerStatusHealthvalueSpec Alpha 255 linear 0.0 0.01
	Animate PlayerStatusHealthvalueSpec FgColor LightGreen accel 0.0 0.0
	Animate PlayerStatusHealthValue Alpha 255 linear 0.0 0.01
	Animate PlayerStatusHealthValue FgColor "Health Numbers" accel 0.0 0.0
	RunEvent HudHealthBonusPulseLoop 0.4
