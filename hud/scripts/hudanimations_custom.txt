
//Health Bonus Pulse
event HudHealthBonusPulse
{
	//Animate	PlayerStatusHealthBonusImage 	Alpha		"255"		Linear 0.0 0.2
	//Animate	PlayerStatusHealthBonusImage 	Alpha		"0"			Linear 0.2 0.4
	
	Animate	test 	Alpha		"0"		Linear 0.0 0.0
	
	Animate	PlayerStatusHealthBonusImage 	Alpha		"0"		Accel 0.0 0.0
	
	Animate PlayerStatusHealthValue FgColor "0 120 201 255" Accel 0.0 0.2
	Animate PlayerStatusHealthValue FgColor "255 255 255 255" Accel 0.2 0.2

	RunEvent HudHealthBonusPulseLoop	0.4
}

// call to loop HudHealthBonusPulse
event HudHealthBonusPulseLoop
{
	RunEvent HudHealthBonusPulse 0.0
}

event HudHealthBonusPulseStop
{
	StopEvent HudHealthBonusPulse 0.0
	StopEvent HudHealthBonusPulseLoop 0.0
	Animate	test 	Alpha		"200"		Linear 0.0 0.0
	
	Animate	PlayerStatusHealthBonusImage 	Alpha		"255"		Accel 0.0 0.0
	Animate PlayerStatusHealthValue FgColor "255 255 255 255" Accel 0.0 0.0
	
}

{

}

{
